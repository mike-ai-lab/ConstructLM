import React, { useState, useEffect, useRef } from 'react';
import { Send, Plus, Mic, Paperclip, ChevronRight, Search } from 'lucide-react';

/**
 * App Component
 * Replicates the KKIA AI Chat interface from the screenshot.
 * Implements a simulated streaming logic for AI responses.
 */
export default function App() {
  const [input, setInput] = useState('');
  const [messages, setMessages] = useState([
    {
      role: 'user',
      content: 'How does the current unsegregated approach to waste collection at KKIA affect operational efficiency and costs?',
      timestamp: '08:20 PM'
    }
  ]);
  const [isStreaming, setIsStreaming] = useState(false);
  const messagesEndRef = useRef(null);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    // Initial simulation of the screenshot's response on mount
    handleSend("How does the current unsegregated approach to waste collection at KKIA affect operational efficiency and costs?", true);
  }, []);

  useEffect(scrollToBottom, [messages, isStreaming]);

  /**
   * Simulated Streaming Logic
   * Animates text character by character to mimic an AI response.
   */
  const streamResponse = async (text) => {
    setIsStreaming(true);
    let displayedText = "";
    
    // Add an empty message placeholder for the AI
    setMessages(prev => [...prev, { role: 'ai', content: '', timestamp: '08:20 PM', isStreaming: true }]);

    const words = text.split(" ");
    for (let i = 0; i < words.length; i++) {
      await new Promise(resolve => setTimeout(resolve, 40)); // Speed of stream
      displayedText += (i === 0 ? "" : " ") + words[i];
      
      setMessages(prev => {
        const newMsgs = [...prev];
        newMsgs[newMsgs.length - 1].content = displayedText;
        return newMsgs;
      });
    }
    
    setMessages(prev => {
      const newMsgs = [...prev];
      newMsgs[newMsgs.length - 1].isStreaming = false;
      return newMsgs;
    });
    setIsStreaming(false);
  };

  const handleSend = async (customText = null, isInitial = false) => {
    const text = customText || input;
    if (!text.trim() || isStreaming) return;

    if (!isInitial) {
      setMessages(prev => [...prev, { role: 'user', content: text, timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }]);
      setInput('');
    }

    const mockResponse = "The current practice of collecting and managing all waste without segregation at KKIA limits recycling potential and increases disposal costs. This unsegregated approach also means that waste is currently collected and managed without segregation.";
    
    await streamResponse(mockResponse);
  };

  return (
    <div className="min-h-screen bg-[#131314] text-[#e3e3e3] font-sans flex flex-col items-center p-4 md:p-8">
      {/* Chat Container */}
      <div className="w-full max-w-3xl flex-1 flex flex-col gap-8 mb-32">
        
        {messages.map((msg, idx) => (
          <div key={idx} className={`flex flex-col ${msg.role === 'user' ? 'items-end' : 'items-start'} animate-in fade-in duration-500`}>
            
            {/* User Message Bubble */}
            {msg.role === 'user' ? (
              <div className="flex items-start gap-3 max-w-[85%]">
                <div className="bg-[#28292a] rounded-[28px] px-6 py-4 text-[15px] leading-relaxed shadow-sm">
                  {msg.content}
                </div>
                <div className="w-8 h-8 rounded-full bg-[#3c3d3e] flex items-center justify-center text-[10px] text-gray-400">
                  U
                </div>
              </div>
            ) : (
              /* AI Response Container */
              <div className="flex flex-col gap-4 w-full">
                <div className="flex items-center gap-3">
                  <div className="w-6 h-6 rounded-full bg-blue-500 flex-shrink-0" />
                  <span className="text-[11px] font-bold tracking-wider text-gray-400 uppercase">Gemini-2.5-Flash-Lite</span>
                </div>

                <div className="pl-9 space-y-4">
                  <p className="text-[17px] leading-[1.6] font-medium text-[#f0f0f0]">
                    {msg.content}
                    {msg.isStreaming && <span className="inline-block w-1.5 h-5 bg-blue-400 ml-1 animate-pulse" />}
                  </p>

                  {!msg.isStreaming && (
                    <div className="pt-4 border-t border-[#3c3d3e] flex flex-col gap-4">
                      <div className="flex items-center gap-2 text-sm text-gray-400">
                        <span>Sources:</span>
                        <span className="text-blue-300 underline cursor-pointer">airport_proposal[1].pdf</span>
                      </div>
                      
                      {/* PDF Attachment UI */}
                      <div className="bg-[#28292a] border border-[#3c3d3e] rounded-xl p-3 flex items-center gap-3 w-fit cursor-pointer hover:bg-[#333537] transition-colors">
                        <div className="bg-[#3c3d3e] p-1.5 rounded">
                           <Paperclip size={14} className="text-gray-300" />
                        </div>
                        <span className="text-xs text-gray-200">airport_proposal[1].pdf</span>
                      </div>
                    </div>
                  )}
                </div>
              </div>
            )}
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      {/* Input Bar Fixed at Bottom */}
      <div className="fixed bottom-8 w-full max-w-3xl px-4">
        <div className="bg-[#1e1f20] rounded-[32px] p-2 flex items-center gap-2 shadow-2xl border border-[#3c3d3e] focus-within:border-[#505050] transition-all">
          <button className="p-3 text-gray-400 hover:text-white transition-colors">
            <Plus size={22} />
          </button>
          
          <input 
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={(e) => e.key === 'Enter' && handleSend()}
            placeholder="Ask anything..." 
            className="flex-1 bg-transparent border-none outline-none text-[#e3e3e3] placeholder-gray-500 py-3 text-[16px]"
          />

          <button className="p-3 text-gray-400 hover:text-white transition-colors">
            <Mic size={22} />
          </button>
          
          {/* Send Button Logic */}
          <button 
            onClick={() => handleSend()}
            disabled={!input.trim() || isStreaming}
            className={`p-3 rounded-full transition-all ${input.trim() ? 'text-blue-400' : 'text-gray-600 cursor-not-allowed'}`}
          >
            <Send size={22} fill={input.trim() ? "currentColor" : "none"} />
          </button>
        </div>
        
        <p className="text-center text-[11px] text-gray-500 mt-4">
          AI can make mistakes. Please verify citations. <span className="opacity-50">v1.0.0-370453</span>
        </p>
      </div>
    </div>
  );
}
